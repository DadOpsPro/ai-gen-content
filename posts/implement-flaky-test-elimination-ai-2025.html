How to Implement Flaky Test Elimination with AI in 2025

Flaky tests are the silent productivity killers lurking in your CI/CD pipeline. They pass one moment, fail the next—without any code changes. In 2025, AI in software testing has evolved beyond simple detection to predictive elimination. This guide walks you through implementing an AI-powered system that identifies, analyzes, and eliminates flaky tests before they waste your team's time.

Why This Matters

Every engineering team knows the frustration: a critical build fails, developers investigate for hours, only to discover the test was simply flaky. The cost is staggering. The 2025 Reality:

  • Teams lose an average of 15-20% of CI/CD compute resources re-running flaky tests
  • Developer productivity drops by 12% when flakiness exceeds 5% of the test suite
  • False confidence from passing flaky tests masks real bugs reaching production

    • Traditional approaches—quarantining tests, adding retries, manual investigation—treat symptoms, not causes. AI-powered flaky test elimination in 2025 addresses the root problem by:

  • Predicting which tests are likely to become flaky before they do
  • Diagnosing the exact cause of flakiness (timing, resource contention, order dependency)
  • Recommending specific fixes with generated code patches
  • Preventing new flaky tests from entering the codebase

    Step 1: Collect and Structure Historical Test Data

    First, extract test execution history from your CI/CD system. This data forms the foundation for AI analysis. 

    test_data_collector.py
    
import requests
from datetime import datetime, timedelta
from dataclasses import dataclass
from typing import List
import json

    

    @dataclass
class TestExecution:
    test_name: str
    run_id: str
    status: str  # 'passed', 'failed', 'skipped'
    duration_ms: float
    timestamp: datetime
    branch: str
    commit_sha: str
    retry_count: int
    error_message: str = None
    
class TestDataCollector:
    def __init__(self, ci_api_url: str, api_token: str):
        self.api_url = ci_api_url
        self.headers = {"Authorization": f"Bearer {api_token}"}
    
    def fetch_github_actions_data(self, repo: str, days: int = 30) -> List[TestExecution]:
        """Fetch test results from GitHub Actions."""
        executions = []
        since = datetime.now() - timedelta(days=days)
        
        # Fetch workflow runs
        runs_url = f"{self.api_url}/repos/{repo}/actions/runs"
        params = {"created": f">={since.isoformat()}", "per_page": 100}
        
        response = requests.get(runs_url, headers=self.headers, params=params)
        runs = response.json().get("workflow_runs", [])
        
        for run in runs:
            # Fetch test artifacts for each run
            artifacts_url = f"{self.api_url}/repos/{repo}/actions/runs/{run['id']}/artifacts"
            artifacts = requests.get(artifacts_url, headers=self.headers).json()
            
            # Parse JUnit XML or JSON test reports
            for artifact in artifacts.get("artifacts", []):
                if "test-results" in artifact["name"]:
                    executions.extend(
                        self._parse_test_artifact(artifact, run)
                    )
        
        return executions
    
    def calculate_flakiness_metrics(self, executions: List[TestExecution]) -> dict:
        """Calculate flakiness score for each test."""
        test_results = {}
        
        for exe in executions:
            if exe.test_name not in test_results:
                test_results[exe.test_name] = {
                    "passes": 0, "failures": 0, "durations": [],
                    "retries": 0, "error_messages": []
                }
            
            result = test_results[exe.test_name]
            if exe.status == "passed":
                result["passes"] += 1
            elif exe.status == "failed":
                result["failures"] += 1
                if exe.error_message:
                    result["error_messages"].append(exe.error_message)
            
            result["durations"].append(exe.duration_ms)
            result["retries"] += exe.retry_count
        
        # Calculate flakiness scores
        metrics = {}
        for test_name, data in test_results.items():
            total = data["passes"] + data["failures"]
            if total < 5:  # Skip tests with insufficient data
                continue
                
            metrics[test_name] = {
                "flakiness_rate": data["failures"] / total if total > 0 else 0,
                "retry_rate": data["retries"] / total,
                "duration_variance": self._calculate_variance(data["durations"]),
                "unique_error_count": len(set(data["error_messages"])),
                "total_runs": total
            }
        
        return metrics

    Step 2: Build the AI Flakiness Detection Model

    Now create a classification model that predicts test flakiness using both statistical features and LLM-based code analysis. 

    flakiness_detector.py
    
import openai
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
import numpy as np
import ast

    

    class AIFlakinessDetector:
    def __init__(self, openai_api_key: str):
        self.client = openai.OpenAI(api_key=openai_api_key)
        self.classifier = RandomForestClassifier(n_estimators=100)
        self.flaky_patterns = []
    
    def extract_code_features(self, test_code: str) -> dict:
        """Use LLM to analyze test code for flakiness indicators."""
        
        prompt = """Analyze this test code for flakiness risk factors. 
Return a JSON object with these scores (0-10):

  • timing_dependency: Uses sleep(), time-based assertions, or timeouts
    • 

  • external_dependency: Calls external APIs, databases, or file systems
    • 

  • state_dependency: Relies on shared state or execution order
    • 

  • resource_contention: May compete for limited resources
    • 

  • non_determinism: Uses random values, current time, or parallel execution
    • 

  • cleanup_issues: Missing or improper test cleanup/teardown
    • 

    

    Test code:
    {code} 
    
Return ONLY valid JSON, no explanation."""

    

            response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": "You are an expert test engineer analyzing code for flakiness risks."},
                {"role": "user", "content": prompt.format(code=test_code)}
            ],
            temperature=0,
            response_format={"type": "json_object"}
        )
        
        return json.loads(response.choices[0].message.content)
    
    def create_feature_vector(self, metrics: dict, code_features: dict) -> np.array:
        """Combine statistical and code-based features."""
        return np.array([
            metrics.get("flakiness_rate", 0),
            metrics.get("retry_rate", 0),
            metrics.get("duration_variance", 0),
            metrics.get("unique_error_count", 0),
            code_features.get("timing_dependency", 0) / 10,
            code_features.get("external_dependency", 0) / 10,
            code_features.get("state_dependency", 0) / 10,
            code_features.get("resource_contention", 0) / 10,
            code_features.get("non_determinism", 0) / 10,
            code_features.get("cleanup_issues", 0) / 10
        ])
    
    def train(self, test_data: List[dict]):
        """Train the flakiness prediction model."""
        X, y = [], []
        
        for test in test_data:
            features = self.create_feature_vector(
                test["metrics"], 
                test["code_features"]
            )
            X.append(features)
            # Label: 1 if historically flaky (>10% failure rate with passes)
            y.append(1 if test["metrics"]["flakiness_rate"] > 0.1 else 0)
        
        X_train, X_test, y_train, y_test = train_test_split(
            np.array(X), np.array(y), test_size=0.2
        )
        
        self.classifier.fit(X_train, y_train)
        accuracy = self.classifier.score(X_test, y_test)
        print(f"Model accuracy: {accuracy:.2%}")
        
        return accuracy
    
    def predict_flakiness(self, test_code: str, metrics: dict = None) -> dict:
        """Predict if a test is or will become flaky."""
        code_features = self.extract_code_features(test_code)
        metrics = metrics or {"flakiness_rate": 0, "retry_rate": 0, 
                             "duration_variance": 0, "unique_error_count": 0}
        
        features = self.create_feature_vector(metrics, code_features)
        probability = self.classifier.predict_proba([features])[0][1]
        
        return {
            "is_flaky": probability > 0.5,
            "flakiness_probability": probability,
            "risk_factors": code_features,
            "recommendation": self._generate_recommendation(code_features)
        }

class FlakyCauseAnalyzer:
    def __init__(self, openai_api_key: str):
        self.client = openai.OpenAI(api_key=openai_api_key)
    
    def analyze_and_fix(self, test_code: str, error_logs: List[str], 
                        test_file_path: str) -> dict:
        """Diagnose flakiness cause and generate a fix."""
        
        analysis_prompt = f"""You are debugging a flaky test. Analyze the test code and error logs to:


  • Identify the ROOT CAUSE of flakiness (not symptoms)
    • 

  • Categorize the cause (timing, state, resource, external, order-dependent)
    • 

  • Generate a FIXED version of the test
    • 

    

    Test code:
    python {test_code} 
    
Recent failure logs (sample of {len(error_logs)} failures):
    {chr(10).join(error_logs[:3])} 
    
Respond in JSON format:
{{
    "root_cause": "detailed explanation",
    "category": "timing|state|resource|external|order",
    "confidence": 0.0-1.0,
    "fixed_code": "complete fixed test code",
    "explanation": "what was changed and why"
}}"""

    

            response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": analysis_prompt}],
            temperature=0,
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        
        # Validate the fix compiles
        try:
            ast.parse(result["fixed_code"])
            result["fix_valid"] = True
        except SyntaxError as e:
            result["fix_valid"] = False
            result["syntax_error"] = str(e)
        
        return result
