    Create a Redis session store middleware: 

    // middleware/session.go
package middleware

    

    import (
    "context"
    "encoding/json"
    "time"
    
    "github.com/redis/go-redis/v9"
    ory "github.com/ory/client-go"
)

    

    type SessionStore struct {
    redis     *redis.ClusterClient
    oryClient *ory.APIClient
    ttl       time.Duration
}

    

    func NewSessionStore(redisAddrs []string, oryURL string) *SessionStore {
    rdb := redis.NewClusterClient(&redis.ClusterOptions{
        Addrs:        redisAddrs,
        PoolSize:     100,
        MinIdleConns: 10,
    })
    
    config := ory.NewConfiguration()
    config.Servers = []ory.ServerConfiguration{{URL: oryURL}}
    
    return &SessionStore{
        redis:     rdb,
        oryClient: ory.NewAPIClient(config),
        ttl:       30  24  time.Hour,
    }
}

    

    func (s SessionStore) ValidateSession(ctx context.Context, sessionToken string) (ory.Session, error) {
    // Check cache first
    cacheKey := "session:" + sessionToken[:16] // Use prefix for security
    
    cached, err := s.redis.Get(ctx, cacheKey).Bytes()
    if err == nil {
        var session ory.Session
        if json.Unmarshal(cached, &session) == nil {
            return &session, nil
        }
    }
    
    // Validate with Ory
    session, _, err := s.oryClient.FrontendAPI.
        ToSession(ctx).
        Cookie("ai_session=" + sessionToken).
        Execute()
    
    if err != nil {
        return nil, err
    }
    
    // Cache the validated session
    sessionBytes, _ := json.Marshal(session)
    s.redis.Set(ctx, cacheKey, sessionBytes, 5*time.Minute)
    
    return session, nil
}

    Step 3: Implement Rate-Limited Authentication Endpoints

    Protect your authentication endpoints from abuse while maintaining performance: 

    // handlers/auth.go
package handlers

    

    import (
    "net/http"
    "time"
    
    "github.com/gin-gonic/gin"
    "golang.org/x/time/rate"
)

    

    type AuthHandler struct {
    sessionStore *middleware.SessionStore
    limiter      *IPRateLimiter
}

    

    type IPRateLimiter struct {
    limiters map[string]*rate.Limiter
    mu       sync.RWMutex
    rate     rate.Limit
    burst    int
}

    

    func NewIPRateLimiter(r rate.Limit, b int) *IPRateLimiter {
    return &IPRateLimiter{
        limiters: make(map[string]*rate.Limiter),
        rate:     r,
        burst:    b,
    }
}

    

    func (l IPRateLimiter) GetLimiter(ip string) rate.Limiter {
    l.mu.Lock()
    defer l.mu.Unlock()
    
    limiter, exists := l.limiters[ip]
    if !exists {
        limiter = rate.NewLimiter(l.rate, l.burst)
        l.limiters[ip] = limiter
    }
    
    return limiter
}

    

    func (h *AuthHandler) AuthMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        // Rate limiting check
        ip := c.ClientIP()
        if !h.limiter.GetLimiter(ip).Allow() {
            c.AbortWithStatusJSON(http.StatusTooManyRequests, gin.H{
                "error": "rate_limit_exceeded",
                "retry_after": 60,
            })
            return
        }
        
        // Session validation
        sessionCookie, err := c.Cookie("ai_session")
        if err != nil {
            c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{
                "error": "session_required",
            })
            return
        }
        
        session, err := h.sessionStore.ValidateSession(c.Request.Context(), sessionCookie)
        if err != nil {
            c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{
                "error": "invalid_session",
            })
            return
        }
        
        // Add user context
        c.Set("user_id", session.Identity.Id)
        c.Set("user_traits", session.Identity.Traits)
        c.Next()
    }
}

Step 5: Implement Token Validation in Your AI Service

ai_service/auth.py

import httpx
from functools import lru_cache
from typing import Optional
import jwt
from pydantic import BaseModel



class TokenClaims(BaseModel):
    sub: str
    scope: str
    exp: int
    iat: int
    client_id: Optional[str] = None



class OryTokenValidator:
    def __init__(self, issuer_url: str, jwks_url: str):
        self.issuer_url = issuer_url
        self.jwks_url = jwks_url
        self._jwks_client = jwt.PyJWKClient(jwks_url, cache_keys=True)
    
    async def validate_access_token(self, token: str) -> TokenClaims:
        """Validate OAuth 2.0 access token issued by Ory Hydra."""
        try:
            signing_key = self._jwks_client.get_signing_key_from_jwt(token)
            
            payload = jwt.decode(
                token,
                signing_key.key,
                algorithms=["RS256"],
                audience="ai-api",
                issuer=self.issuer_url,
                options={"require": ["exp", "sub", "scope"]}
            )
            
            return TokenClaims(**payload)
            
        except jwt.ExpiredSignatureError:
            raise AuthenticationError("Token expired")
        except jwt.InvalidTokenError as e:
            raise AuthenticationError(f"Invalid token: {e}")
    
    def has_scope(self, claims: TokenClaims, required_scope: str) -> bool:
        """Check if token has required scope."""
        scopes = claims.scope.split(" ")
        return required_scope in scopes


FastAPI integration

from fastapi import Depends, HTTPException, Security
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials



security = HTTPBearer()
validator = OryTokenValidator(
    issuer_url="https://auth.yourdomain.com/",
    jwks_url="https://auth.yourdomain.com/.well-known/jwks.json"
)



async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Security(security)
) -> TokenClaims:
    try:
        claims = await validator.validate_access_token(credentials.credentials)
        return claims
    except AuthenticationError as e:
        raise HTTPException(status_code=401, detail=str(e))



@app.post("/v1/chat/completions")
async def chat_completions(
    request: ChatRequest,
    user: TokenClaims = Depends(get_current_user)
):
    if not validator.has_scope(user, "ai:write"):
        raise HTTPException(status_code=403, detail="Insufficient scope")
    
    # Process AI request with user context
    return await process_chat(request, user_id=user.sub)

Common Pitfalls & How to Avoid Them

Pitfall 1: Synchronous Session Validation

Problem: Validating sessions against Ory on every request creates latency spikes. Solution: Implement Redis caching with 5-minute TTL as shown in Step 2. Use Redis cluster mode for high availability.

Pitfall 2: Single Point of Failure

Problem: Running single instances of Ory components causes outages during updates. Solution: Always deploy minimum 3 replicas with pod disruption budgets: 
podDisruptionBudget:
  enabled: true
  minAvailable: 2
