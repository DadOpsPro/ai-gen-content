    Step 1: Audit Your Current Test Suite

    Before introducing AI tools, understand your baseline. Run this analysis script against your existing test repository: 

    test_suite_analyzer.py
    
import os
import re
from pathlib import Path
from collections import defaultdict

    

    def analyze_test_suite(test_directory: str) -> dict:
    """Analyze existing tests for AI tool compatibility."""
    
    metrics = {
        "total_tests": 0,
        "flaky_indicators": 0,
        "hardcoded_selectors": 0,
        "sleep_statements": 0,
        "files_by_type": defaultdict(int)
    }
    
    flaky_patterns = [
        r'sleep\(\d+\)',
        r'wait\(\d+\)',
        r'retry.*count',
        r'flaky|unstable|skip'
    ]
    
    selector_patterns = [
        r'#[a-zA-Z][\w-]+',  # ID selectors
        r'\[data-testid=["\'][^"\']+["\']\]',
        r'\.[\w-]+',  # Class selectors
        r'xpath\s=\s["\']'
    ]
    
    for filepath in Path(test_directory).rglob("test.py"):
        metrics["total_tests"] += 1
        metrics["files_by_type"]["python"] += 1
        
        content = filepath.read_text()
        
        for pattern in flaky_patterns:
            metrics["flaky_indicators"] += len(re.findall(pattern, content, re.I))
        
        for pattern in selector_patterns:
            metrics["hardcoded_selectors"] += len(re.findall(pattern, content))
    
    # Calculate health score
    total_issues = metrics["flaky_indicators"] + metrics["hardcoded_selectors"]
    metrics["health_score"] = max(0, 100 - (total_issues / max(metrics["total_tests"], 1) * 10))
    
    return metrics

    

    if __name__ == "__main__":
    results = analyze_test_suite("./tests")
    print(f"Test Suite Health Score: {results['health_score']:.1f}/100")
    print(f"Total Tests: {results['total_tests']}")
    print(f"Flaky Indicators Found: {results['flaky_indicators']}")
    print(f"Hardcoded Selectors: {results['hardcoded_selectors']}")

    // lib/ai-locator.js
const { chromium } = require('@playwright/test');
const axios = require('axios');
const config = require('../healenium.config');

    

    class AILocator {
    constructor(page) {
        this.page = page;
        this.healingHistory = [];
    }

    

        async findElement(selector, options = {}) {
        const { timeout = 5000, healOnFailure = true } = options;
        
        try {
            // Attempt standard location
            const element = await this.page.waitForSelector(selector, { timeout });
            return element;
        } catch (error) {
            if (!healOnFailure || !config.healEnabled) {
                throw error;
            }
            
            // Trigger AI healing
            return await this.healAndFind(selector, error);
        }
    }

    

        async healAndFind(originalSelector, originalError) {
        console.log(🔧 AI Healing triggered for: ${originalSelector});
        
        // Capture page state for analysis
        const pageHtml = await this.page.content();
        const screenshot = await this.page.screenshot({ encoding: 'base64' });
        
        try {
            // Send to Healenium server for analysis
            const response = await axios.post(${config.serverUrl}/heal, {
                originalSelector,
                pageContent: pageHtml,
                screenshot,
                url: this.page.url()
            });
            
            if (response.data.healedSelector && response.data.confidence >= config.scoreCap) {
                const healedSelector = response.data.healedSelector;
                console.log(✅ Healed selector: ${healedSelector} (confidence: ${response.data.confidence}));
                
                // Log healing for later review
                this.healingHistory.push({
                    timestamp: new Date().toISOString(),
                    original: originalSelector,
                    healed: healedSelector,
                    confidence: response.data.confidence,
                    url: this.page.url()
                });
                
                return await this.page.waitForSelector(healedSelector, { timeout: 5000 });
            }
        } catch (healError) {
            console.error('Healing service error:', healError.message);
        }
        
        throw originalError;
    }

    

        getHealingReport() {
        return this.healingHistory;
    }
}

    

    module.exports = { AILocator };

const { chromium } = require('@playwright/test');

    class AITestGenerator {

        async recordUserJourney(journeyName) {
        const browser = await chromium.launch({ headless: false });
        const context = await browser.newContext({
            recordHar: { path: ./recordings/${journeyName}.har }
        });
        const page = await context.newPage();
        
        const actions = [];
        
        // Intercept and record all user actions
        page.on('click', async (element) => {
            actions.push({
                type: 'click',
                selector: await this.generateRobustSelector(element),
                timestamp: Date.now()
            });
        });
        
        page.on('fill', async ({ element, value }) => {
            actions.push({
                type: 'fill',
                selector: await this.generateRobustSelector(element),
                value: this.sanitizeValue(value),
                timestamp: Date.now()
            });
        });

    

            await page.goto(this.baseUrl);
        
        console.log(\n🎬 Recording started for: ${journeyName});
        console.log('Perform your user journey, then press Ctrl+C to stop.\n');
        
        // Wait for user to complete journey
        await new Promise((resolve) => {
            process.on('SIGINT', async () => {
                await this.generateTest(journeyName, actions);
                await browser.close();
                resolve();
            });
        });
    }

    

        async generateTest(journeyName, actions) {
        const testCode = this.buildTestCode(journeyName, actions);
        
        const outputPath = path.join(this.outputDir, ${journeyName}.spec.js);
        await fs.mkdir(this.outputDir, { recursive: true });
        await fs.writeFile(outputPath, testCode);
        
        console.log(\n✅ Test generated: ${outputPath});
    }

    

        buildTestCode(journeyName, actions) {
        const actionCode = actions.map(action => {
            switch (action.type) {
                case 'click':
                    return     await aiLocator.findElement('${action.selector}').then(el => el.click());;
                case 'fill':
                    return     await aiLocator.findElement('${action.selector}').then(el => el.fill('${action.value}'));;
                default:
                    return '';
            }
        }).join('\n');

${actionCode}

        generateRobustSelector(element) {
        // Priority: data-testid > aria-label > id > CSS path
        const testId = element.getAttribute('data-testid');
        if (testId) return [data-testid="${testId}"];
        
        const ariaLabel = element.getAttribute('aria-label');
        if (ariaLabel) return [aria-label="${ariaLabel}"];
        
        const id = element.getAttribute('id');
        if (id) return #${id};
        
        // Fallback to role-based selector
        const role = element.getAttribute('role') || element.tagName.toLowerCase();
        const text = element.textContent?.trim().slice(0, 30);
        return ${role}:has-text("${text}");
    }

    

        sanitizeValue(value) {
        // Replace sensitive data with test placeholders
        if (value.includes('@')) return 'test@example.com';
        if (/^\d{10,}$/.test(value)) return '1234567890';
        return value;
    }
}

Common Pitfalls & How to Avoid Them

Pitfall 1: Over-reliance on AI healing Problem: Teams accept all healed selectors without review, masking underlying UI instability. Solution: Implement weekly healing report reviews. If the same element heals repeatedly, update the source selector or add proper data-testid attributes. Pitfall 2: Training AI on flaky tests Problem: AI tools learn from your existing tests—garbage in, garbage out. Solution: Clean your test suite before AI adoption. Remove tests with sleep() statements and stabilize flaky tests first.
