How to Implement Flaky Test Elimination with AI in 2025

Flaky tests—those inconsistent tests that pass and fail without code changes—remain one of the most frustrating challenges in software testing. In 2025, AI in Software Testing has matured to the point where we can systematically identify, analyze, and eliminate these unreliable tests. This guide walks you through implementing an AI-powered flaky test elimination pipeline that actually works.

Why This Matters

Flaky tests are more than just annoying—they're expensive. According to recent industry data, engineering teams spend an average of 15-20% of their CI/CD time dealing with flaky test failures. The hidden costs compound quickly:

  • Developer productivity loss: Engineers waste hours investigating failures that aren't real bugs
  • Eroded trust: Teams start ignoring test failures, missing actual regressions
  • Slower releases: Flaky tests block deployments and extend release cycles
  • Technical debt accumulation: Disabled or quarantined tests pile up without resolution

    • Traditional approaches—rerunning tests, adding arbitrary waits, or simply deleting problematic tests—don't address root causes. AI-powered flaky test elimination changes this by automatically detecting patterns, identifying root causes, and suggesting targeted fixes.

Step 1: Instrument Your Test Suite for Data Collection

Before AI can help, you need quality data. Instrument your tests to capture detailed execution metrics beyond simple pass/fail.

Create a custom test reporter that captures timing, environment, and execution context: 

flaky_detector/collector.py

import json
import time
import platform
import psutil
from datetime import datetime
from dataclasses import dataclass, asdict
from typing import Optional



@dataclass
class TestExecutionRecord:
    test_id: str
    test_name: str
    suite_name: str
    outcome: str  # 'passed', 'failed', 'skipped', 'error'
    duration_ms: float
    timestamp: str
    run_id: str
    retry_number: int
    error_message: Optional[str]
    error_type: Optional[str]
    
    # Environment context
    cpu_percent: float
    memory_percent: float
    hostname: str
    python_version: str
    
    # Test context
    test_file: str
    test_line_number: int
    dependencies: list  # Other tests that ran before this
    
class FlakyTestCollector:
    def __init__(self, output_path: str, run_id: str):
        self.output_path = output_path
        self.run_id = run_id
        self.records = []
        self.execution_order = []
    
    def record_test(self, test_item, outcome, duration, error=None):
        record = TestExecutionRecord(
            test_id=f"{test_item.module.__name__}::{test_item.name}",
            test_name=test_item.name,
            suite_name=test_item.module.__name__,
            outcome=outcome,
            duration_ms=duration * 1000,
            timestamp=datetime.utcnow().isoformat(),
            run_id=self.run_id,
            retry_number=getattr(test_item, '_retry_count', 0),
            error_message=str(error) if error else None,
            error_type=type(error).__name__ if error else None,
            cpu_percent=psutil.cpu_percent(),
            memory_percent=psutil.virtual_memory().percent,
            hostname=platform.node(),
            python_version=platform.python_version(),
            test_file=str(test_item.fspath),
            test_line_number=test_item.reportinfo()[1],
            dependencies=self.execution_order.copy()
        )
        self.records.append(asdict(record))
        self.execution_order.append(record.test_id)
    
    def flush(self):
        with open(self.output_path, 'w') as f:
            json.dump(self.records, f, indent=2)

conftest.py

import pytest
import uuid
from flaky_detector.collector import FlakyTestCollector



collector = None



def pytest_configure(config):
    global collector
    run_id = str(uuid.uuid4())[:8]
    collector = FlakyTestCollector(f"test_results_{run_id}.json", run_id)



def pytest_runtest_makereport(item, call):
    if call.when == "call":
        outcome = "passed" if call.excinfo is None else "failed"
        error = call.excinfo.value if call.excinfo else None
        collector.record_test(item, outcome, call.duration, error)



def pytest_sessionfinish(session, exitstatus):
    collector.flush()

def init_database(db_path: str = "flaky_tests.db"):

            test_id TEXT NOT NULL,

            suite_name TEXT NOT NULL,

            duration_ms REAL,

            run_id TEXT,

            error_type TEXT,

            memory_percent REAL,
            hostname TEXT,
            test_file TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    ''')
    
    cursor.execute('''
        CREATE INDEX IF NOT EXISTS idx_test_id ON test_executions(test_id)
    ''')
    
    cursor.execute('''
        CREATE INDEX IF NOT EXISTS idx_timestamp ON test_executions(timestamp)
    ''')
    
    conn.commit()
    return conn



def import_results(conn, json_file: str):
    with open(json_file) as f:
        records = json.load(f)
    
    cursor = conn.cursor()
    for record in records:
        cursor.execute('''
            INSERT INTO test_executions 
            (test_id, test_name, suite_name, outcome, duration_ms, timestamp,
             run_id, retry_number, error_message, error_type, cpu_percent,

            record['outcome'], record['duration_ms'], record['timestamp'],
            record['run_id'], record['retry_number'], record['error_message'],
            record['error_type'], record['cpu_percent'], record['memory_percent'],
            record['hostname'], record['test_file']
        ))
    conn.commit()

@dataclass class FlakyTestReport: test_id: str flakiness_score: float # 0-1, higher = more flaky failure_rate: float variance_score: float predicted_cause: str confidence: float recommended_fix: str evidence: dict

class AIFlakyDetector: # Common flaky test patterns and their signatures CAUSE_PATTERNS = { 'timing_dependency': { 'indicators': ['duration_variance_high', 'timeout_errors'], 'fix': 'Add explicit waits or polling mechanisms' }, 'resource_contention': { 'indicators': ['high_cpu_correlation', 'parallel_failures'], 'fix': 'Add resource isolation or sequential execution' }, 'test_order_dependency': { 'indicators': ['position_correlation', 'shared_state_errors'], 'fix': 'Add proper setup/teardown, isolate test state' }, 'external_service': { 'indicators': ['network_errors', 'time_of_day_pattern'], 'fix': 'Add mocks/stubs for external dependencies' }, 'race_condition': { 'indicators': ['async_errors', 'inconsistent_state'], 'fix': 'Add synchronization primitives or async/await patterns' } } def __init__(self, db_path: str): self.db_path = db_path self.scaler = StandardScaler() self.anomaly_detector = IsolatedForest(contamination=0.1, random_state=42) self.cause_classifier = RandomForestClassifier(n_estimators=100, random_state=42) def load_test_data(self, test_id: str) -> pd.DataFrame: conn = sqlite3.connect(self.db_path) query = ''' SELECT * FROM test_executions WHERE test_id = ? ORDER BY timestamp DESC LIMIT 500 ''' df = pd.read_sql_query(query, conn, params=(test_id,)) conn.close() return df def calculate_flakiness_score(self, df: pd.DataFrame) -> Tuple[float, dict]: if len(df) < 10: return 0.0, {'reason': 'insufficient_data'} # Core metrics failure_rate = (df['outcome'] == 'failed').mean() # A test is flaky if it sometimes passes and sometimes fails outcomes = df['outcome'].value_counts(normalize=True) has_mixed_outcomes = len(outcomes) > 1 and outcomes.min() > 0.05 # Duration variance (normalized) duration_cv = df['duration_ms'].std() / df['duration_ms'].mean() if df['duration_ms'].mean() > 0 else 0 # Flip frequency (how often does outcome change between runs?) df_sorted = df.sort_values('timestamp') outcome_changes = (df_sorted['outcome'] != df_sorted['outcome'].shift()).sum() flip_rate = outcome_changes / len(df) if len(df) > 1 else 0 # Calculate composite score if not has_mixed_outcomes: flakiness_score = 0.0 # Consistently passing or failing = not flaky else: flakiness_score = ( 0.4 min(failure_rate 2, 1.0) + # Moderate failure rate 0.3 * flip_rate + # Frequent outcome changes 0.3 * min(duration_cv, 1.0) # Timing instability ) evidence = { 'failure_rate': failure_rate, 'flip_rate': flip_rate, 'duration_cv': duration_cv, 'sample_size': len(df), 'has_mixed_outcomes': has_mixed_outcomes } return min(flakiness_score, 1.0), evidence def predict_root_cause(self, df: pd.DataFrame) -> Tuple[str, float]: if len(df) < 20: return 'unknown', 0.0 indicators = [] # Check timing patterns failed_runs = df[df['outcome'] == 'failed'] passed_runs = df[df['outcome'] == 'passed'] if len(failed_runs) > 0 and len(passed_runs) > 0: # Duration difference between pass/fail duration_diff = abs(failed_runs['duration_ms'].mean() - passed_runs['duration_ms'].mean()) if duration_diff > passed_runs['duration_ms'].std() * 2: indicators.append('duration_variance_high') # CPU correlation if failed_runs['cpu_percent'].mean() > passed_runs['cpu_percent'].mean() * 1.2: indicators.append('high_cpu_correlation') # Error message analysis if failed_runs['error_type'].notna().any(): error_types = failed_runs['error_type'].value_counts() if 'TimeoutError' in error_types.index or 'AsyncioTimeoutError' in error_types.index: indicators.append('timeout_errors') if 'ConnectionError' in error_types.index or 'HTTPError' in error_types.index: indicators.append('network_errors') # Match to known patterns best_match = 'unknown' best_score = 0.0 for cause, pattern in self.CAUSE_PATTERNS.items(): match_count = sum(1 for ind in indicators if ind in pattern['indicators']) score = match_count / len(pattern['indicators']) if pattern['indicators'] else 0 if score > best_score: best_score = score best_match = cause return best_match, best_score def analyze_all_tests(self, min_executions: int = 20) -> List[FlakyTestReport]: conn = sqlite3.connect(self.db_path) # Get all tests with sufficient history query = ''' SELECT test_id, COUNT(*) as exec_count FROM test_executions GROUP BY test_id HAVING exec_count >= ? ''' tests = pd.read_sql_query(query, conn, params=(min_executions,)) conn.close() reports = [] for _, row in tests.iterrows(): test_id = row['test_id'] df = self.load_test_data(test_id) flakiness_score, evidence = self.calculate_flakiness_score(df) if flakiness_score > 0.1: # Threshold for "flaky" cause, confidence = self.predict_root_cause(df) fix = self.CAUSE_PATTERNS.get(cause, {}).get('fix', 'Manual investigation required') reports.append(FlakyTestReport( test_id=test_id, flakiness_score=flakiness_score, failure_rate=evidence['failure_rate'], variance_score=evidence['duration_cv'], predicted_cause=cause, confidence=confidence, recommended_fix=fix, evidence=evidence )) # Sort by flakiness score descending reports.
