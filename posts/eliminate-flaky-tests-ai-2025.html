How to Implement Flaky Test Elimination with AI in 2025

Why This Matters

Flaky tests are the silent productivity killers in modern software development. These tests pass and fail intermittently without any code changes, eroding developer trust in test suites and wasting countless engineering hours. According to recent industry data, teams spend up to 25% of their CI/CD time dealing with flaky tests, and the average enterprise codebase contains 15-25% flaky tests.

In 2025, AI in Software Testing has matured beyond basic test generation to tackle one of the most frustrating problems in quality assurance: flaky test identification and elimination. Traditional approaches—manual investigation, quarantine-and-forget strategies, and brute-force reruns—no longer scale for modern development velocity. Flaky test elimination with AI leverages machine learning models trained on test execution patterns, code changes, environmental signals, and historical failure data to automatically detect, diagnose, and remediate flaky tests. This guide provides a concrete implementation path for integrating AI-powered flaky test elimination into your existing test infrastructure.

    Step-by-Step Instructions

    Step 1: Instrument Your Test Framework for Rich Data Collection

    AI models require comprehensive data to identify flaky patterns. Start by instrumenting your test framework to capture execution context.

    For pytest, create a custom plugin that collects execution metadata: 

    conftest.py - Flaky Test Data Collection Plugin
    
import pytest
import json
import time
import psutil
import hashlib
from datetime import datetime
from pathlib import Path

    

    class FlakyTestCollector:
    def __init__(self):
        self.test_data = []
        self.session_id = hashlib.md5(
            str(datetime.now().timestamp()).encode()
        ).hexdigest()[:12]
    
    def collect_environment_snapshot(self):
        return {
            "cpu_percent": psutil.cpu_percent(interval=0.1),
            "memory_percent": psutil.virtual_memory().percent,
            "disk_io_read": psutil.disk_io_counters().read_bytes,
            "timestamp": datetime.utcnow().isoformat(),
            "active_threads": len(psutil.Process().threads())
        }

    

    collector = FlakyTestCollector()

    

    @pytest.hookimpl(tryfirst=True)
def pytest_runtest_protocol(item, nextitem):
    item._flaky_start_time = time.perf_counter()
    item._flaky_env_before = collector.collect_environment_snapshot()
    return None

    

    @pytest.hookimpl(trylast=True)
def pytest_runtest_makereport(item, call):
    if call.when == "call":
        duration = time.perf_counter() - item._flaky_start_time
        env_after = collector.collect_environment_snapshot()
        
        test_record = {
            "session_id": collector.session_id,
            "test_id": item.nodeid,
            "test_file": str(item.fspath),
            "outcome": "passed" if call.excinfo is None else "failed",
            "duration_seconds": duration,
            "exception_type": str(call.excinfo.typename) if call.excinfo else None,
            "exception_message": str(call.excinfo.value)[:500] if call.excinfo else None,
            "env_before": item._flaky_env_before,
            "env_after": env_after,
            "markers": [m.name for m in item.iter_markers()],
            "git_commit": os.environ.get("GIT_COMMIT", "unknown"),
            "ci_build_id": os.environ.get("BUILD_ID", "local")
        }
        collector.test_data.append(test_record)

    

    @pytest.hookimpl(trylast=True)
def pytest_sessionfinish(session, exitstatus):
    output_path = Path("test_execution_data") / f"{collector.session_id}.json"
    output_path.parent.mkdir(exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(collector.test_data, f, indent=2)

    This instrumentation captures timing, resource utilization, and failure context—all critical signals for AI-based flaky detection.

    Step 2: Build Your Historical Test Execution Database

    Aggregate collected data into a queryable database. TimescaleDB or ClickHouse work well for time-series test data. 

    flaky_data_pipeline.py
    
import pandas as pd
from sqlalchemy import create_engine, text
from pathlib import Path
import json

    

    class FlakyDataPipeline:
    def __init__(self, db_connection_string):
        self.engine = create_engine(db_connection_string)
        self._initialize_schema()
    
    def _initialize_schema(self):
        schema = """
        CREATE TABLE IF NOT EXISTS test_executions (
            id SERIAL PRIMARY KEY,
            session_id VARCHAR(12),
            test_id VARCHAR(500),
            outcome VARCHAR(20),
            duration_seconds FLOAT,
            exception_type VARCHAR(200),
            cpu_before FLOAT,
            cpu_after FLOAT,
            memory_before FLOAT,
            memory_after FLOAT,
            git_commit VARCHAR(40),
            executed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            INDEX idx_test_id (test_id),
            INDEX idx_executed_at (executed_at)
        );
        """
        with self.engine.connect() as conn:
            conn.execute(text(schema))
            conn.commit()
    
    def ingest_execution_files(self, data_directory: Path):
        for json_file in data_directory.glob("*.json"):
            with open(json_file) as f:
                records = json.load(f)
            
            df = pd.json_normalize(records)
            df = df.rename(columns={
                "env_before.cpu_percent": "cpu_before",
                "env_after.cpu_percent": "cpu_after",
                "env_before.memory_percent": "memory_before",
                "env_after.memory_percent": "memory_after"
            })
            
            df.to_sql("test_executions", self.engine, 
                     if_exists="append", index=False)
            json_file.unlink()  # Remove processed file

    Step 3: Implement AI-Powered Flaky Detection Model

    With sufficient historical data (minimum 100 executions per test), train a detection model. This example uses scikit-learn for a production-ready classifier: 

    flaky_detector.py
    
import pandas as pd
import numpy as np
from sklearn.ensemble import IsolationForest, RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from sqlalchemy import create_engine, text
from dataclasses import dataclass
from typing import List, Tuple

    

    @dataclass
class FlakyTestResult:
    test_id: str
    flakiness_score: float
    flaky_pattern: str
    confidence: float
    recommended_action: str

    

    class AIFlakyDetector:
    def __init__(self, db_connection_string: str):
        self.engine = create_engine(db_connection_string)
        self.scaler = StandardScaler()
        self.isolation_model = IsolationForest(
            contamination=0.1, 
            random_state=42,
            n_estimators=200
        )
        
    def extract_test_features(self, test_id: str) -> pd.DataFrame:
        query = """
        SELECT 
            test_id,
            outcome,
            duration_seconds,
            cpu_before,
            cpu_after,
            memory_before,
            memory_after,
            exception_type,
            executed_at
        FROM test_executions
        WHERE test_id = :test_id
        AND executed_at > NOW() - INTERVAL '30 days'
        ORDER BY executed_at
        """
        
        with self.engine.connect() as conn:
            df = pd.read_sql(text(query), conn, params={"test_id": test_id})
        
        if len(df) < 10:
            return None
            
        # Calculate flakiness indicators
        features = {
            "test_id": test_id,
            "total_runs": len(df),
            "failure_rate": (df["outcome"] == "failed").mean(),
            "pass_fail_transitions": self._count_transitions(df["outcome"]),
            "duration_cv": df["duration_seconds"].std() / df["duration_seconds"].mean(),
            "duration_trend": self._calculate_trend(df["duration_seconds"]),
            "cpu_correlation": df["cpu_before"].corr(
                (df["outcome"] == "failed").astype(int)
            ),
            "memory_correlation": df["memory_before"].corr(
                (df["outcome"] == "failed").astype(int)
            ),
            "unique_exceptions": df["exception_type"].nunique(),
            "time_of_day_variance": self._time_variance(df["executed_at"])
        }
        
        return pd.DataFrame([features])
    
    def _count_transitions(self, outcomes: pd.Series) -> int:
        return (outcomes != outcomes.shift()).sum() - 1
    
    def _calculate_trend(self, series: pd.Series) -> float:
        if len(series) < 2:
            return 0
        x = np.arange(len(series))
        return np.polyfit(x, series.values, 1)[0]
    
    def _time_variance(self, timestamps: pd.Series) -> float:
        hours = pd.to_datetime(timestamps).dt.hour
        return hours.std()
    
    def detect_flaky_tests(self, min_runs: int = 20) -> List[FlakyTestResult]:
        # Get all tests with sufficient execution history
        query = """
        SELECT DISTINCT test_id 
        FROM test_executions
        WHERE executed_at > NOW() - INTERVAL '30 days'
        GROUP BY test_id
        HAVING COUNT(*) >= :min_runs
        """
        
        with self.engine.connect() as conn:
            test_ids = pd.read_sql(
                text(query), conn, params={"min_runs": min_runs}
            )["test_id"].tolist()
        
        # Extract features for all tests
        all_features = []
        for test_id in test_ids:
            features = self.extract_test_features(test_id)
            if features is not None:
                all_features.append(features)
        
        if not all_features:
            return []
            
        features_df = pd.concat(all_features, ignore_index=True)
        
        # Prepare numeric features for anomaly detection
        numeric_cols = [
            "failure_rate", "pass_fail_transitions", "duration_cv",
            "duration_trend", "cpu_correlation", "memory_correlation",
            "unique_exceptions", "time_of_day_variance"
        ]
        
        X = features_df[numeric_cols].fillna(0)
        X_scaled = self.scaler.fit_transform(X)
        
        # Detect anomalies (flaky tests exhibit unusual patterns)
        anomaly_scores = self.isolation_model.fit_predict(X_scaled)
        features_df["anomaly_score"] = self.isolation_model.decision_function(X_scaled)
        
        # Classify flaky patterns and generate recommendations
        results = []
        for _, row in features_df.iterrows():
            flakiness_score = self._calculate_flakiness_score(row)
            
            if flakiness_score > 0.3:  # Threshold for flagging
                pattern, action = self._classify_pattern(row)
                results.append(FlakyTestResult(
                    test_id=row["test_id"],
                    flakiness_score=flakiness_score,
                    flaky_pattern=pattern,
                    confidence=min(0.95, row["total_runs"] / 100),
                    recommended_action=action
                ))
        
        return sorted(results, key=lambda x: x.flakiness_score, reverse=True)
    
    def _calculate_flakiness_score(self, row: pd.Series) -> float:
        # Weighted combination of flakiness indicators
        score = (
            0.3 * min(1.0, row["pass_fail_transitions"] / 10) +
            0.25  row["failure_rate"]  (1 - row["failure_rate"]) * 4 +
            0.2 * min(1.0, row["duration_cv"]) +
            0.15 * min(1.0, abs(row["cpu_correlation"])) +
            0.1 * min(1.0, row["unique_exceptions"] / 3)
        )
        return round(score, 3)
    
    def _classify_pattern(self, row: pd.Series) -> Tuple[str, str]:
        if abs(row["cpu_correlation"]) > 0.5:
            return "resource_contention", "Add resource isolation or increase timeouts"
        elif row["duration_cv"] > 0.5:
            return "timing_dependency", "Review async operations and add explicit waits"
        elif row["unique_exceptions"] > 2:
            return "environment_variance", "Standardize test environment configuration"
        elif row["pass_fail_transitions"] > row["total_runs"] * 0.3:
            return "order_dependency", "Check for shared state between tests"
        else:
            return "unknown_pattern", "Manual investigation recommended"
