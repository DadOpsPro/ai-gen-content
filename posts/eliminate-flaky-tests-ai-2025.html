    Step 1: Collect and Structure Historical Test Data

    AI models require historical context to identify flakiness patterns. Start by capturing comprehensive test execution metadata.

    Create a data collection script that runs after each test execution: 

    flaky_data_collector.py
    
import json
import xml.etree.ElementTree as ET
from datetime import datetime
from dataclasses import dataclass, asdict
import hashlib
import psycopg2

    

    @dataclass
class TestExecution:
    test_id: str
    test_name: str
    test_file: str
    status: str  # passed, failed, skipped
    duration_ms: float
    error_message: str | None
    stack_trace: str | None
    commit_sha: str
    branch: str
    runner_id: str
    execution_timestamp: str
    retry_count: int
    environment_hash: str

    

    def parse_junit_xml(xml_path: str, build_context: dict) -> list[TestExecution]:
    """Parse JUnit XML and extract test execution data."""
    tree = ET.parse(xml_path)
    root = tree.getroot()
    executions = []
    
    for testsuite in root.findall('.//testsuite'):
        for testcase in testsuite.findall('testcase'):
            test_name = testcase.get('name')
            test_file = testcase.get('file', testsuite.get('name', 'unknown'))
            
            # Determine status
            failure = testcase.find('failure')
            error = testcase.find('error')
            skipped = testcase.find('skipped')
            
            if failure is not None:
                status = 'failed'
                error_message = failure.get('message', '')
                stack_trace = failure.text
            elif error is not None:
                status = 'failed'
                error_message = error.get('message', '')
                stack_trace = error.text
            elif skipped is not None:
                status = 'skipped'
                error_message = None
                stack_trace = None
            else:
                status = 'passed'
                error_message = None
                stack_trace = None
            
            # Generate stable test ID
            test_id = hashlib.sha256(
                f"{test_file}::{test_name}".encode()
            ).hexdigest()[:16]
            
            executions.append(TestExecution(
                test_id=test_id,
                test_name=test_name,
                test_file=test_file,
                status=status,
                duration_ms=float(testcase.get('time', 0)) * 1000,
                error_message=error_message,
                stack_trace=stack_trace,
                commit_sha=build_context['commit_sha'],
                branch=build_context['branch'],
                runner_id=build_context['runner_id'],
                execution_timestamp=datetime.utcnow().isoformat(),
                retry_count=build_context.get('retry_count', 0),
                environment_hash=build_context['env_hash']
            ))
    
    return executions

    

    def store_executions(executions: list[TestExecution], db_conn_string: str):
    """Store test executions in PostgreSQL for AI analysis."""
    conn = psycopg2.connect(db_conn_string)
    cursor = conn.cursor()
    
    insert_query = """
        INSERT INTO test_executions 
        (test_id, test_name, test_file, status, duration_ms, 
         error_message, stack_trace, commit_sha, branch, runner_id,
         execution_timestamp, retry_count, environment_hash)
        VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
    """
    
    for exec in executions:
        cursor.execute(insert_query, (
            exec.test_id, exec.test_name, exec.test_file, exec.status,
            exec.duration_ms, exec.error_message, exec.stack_trace,
            exec.commit_sha, exec.branch, exec.runner_id,
            exec.execution_timestamp, exec.retry_count, exec.environment_hash
        ))
    
    conn.commit()
    cursor.close()
    conn.close()

    Step 2: Implement Flakiness Detection Algorithm

    Deploy an AI model that analyzes execution patterns to classify test reliability. Launchable and BuildPulse offer pre-trained models, but here's how to implement a custom solution: 

    flaky_detector.py
    
import pandas as pd
import numpy as np
from sklearn.ensemble import IsolationForest
from dataclasses import dataclass

    

    @dataclass
class FlakinessPrediction:
    test_id: str
    flakiness_score: float  # 0-1, higher = more flaky
    confidence: float
    primary_pattern: str
    recommended_action: str

    

    class AIFlakyDetector:
    def __init__(self, db_conn_string: str):
        self.db_conn = db_conn_string
        self.model = IsolationForest(contamination=0.1, random_state=42)
        
    def compute_test_features(self, test_id: str, lookback_days: int = 30) -> dict:
        """Extract features for flakiness prediction."""
        query = f"""
            SELECT status, duration_ms, error_message, branch,
                   execution_timestamp, retry_count
            FROM test_executions
            WHERE test_id = %s
              AND execution_timestamp > NOW() - INTERVAL '{lookback_days} days'
            ORDER BY execution_timestamp
        """
        df = pd.read_sql(query, self.db_conn, params=(test_id,))
        
        if len(df) < 10:
            return None  # Insufficient data
        
        # Core flakiness indicators
        pass_count = (df['status'] == 'passed').sum()
        fail_count = (df['status'] == 'failed').sum()
        total = pass_count + fail_count
        
        # Flip rate: how often does status change between consecutive runs?
        df['status_changed'] = df['status'] != df['status'].shift(1)
        flip_rate = df['status_changed'].sum() / len(df)
        
        # Same-commit variance: different results for identical code
        commit_variance = df.groupby('branch').apply(
            lambda x: x['status'].nunique() > 1
        ).mean()
        
        # Duration instability
        duration_cv = df['duration_ms'].std() / df['duration_ms'].mean()
        
        # Error message diversity (multiple failure modes = likely flaky)
        unique_errors = df[df['error_message'].notna()]['error_message'].nunique()
        
        # Retry success rate
        retry_df = df[df['retry_count'] > 0]
        retry_success = (retry_df['status'] == 'passed').mean() if len(retry_df) > 0 else 0
        
        return {
            'test_id': test_id,
            'pass_rate': pass_count / total if total > 0 else 1,
            'flip_rate': flip_rate,
            'commit_variance': commit_variance,
            'duration_cv': duration_cv,
            'unique_error_count': unique_errors,
            'retry_success_rate': retry_success,
            'execution_count': total
        }
    
    def predict_flakiness(self, test_id: str) -> FlakinessPrediction:
        """Generate flakiness prediction with actionable recommendations."""
        features = self.compute_test_features(test_id)
        if features is None:
            return FlakinessPrediction(
                test_id=test_id,
                flakiness_score=0.0,
                confidence=0.0,
                primary_pattern="insufficient_data",
                recommended_action="collect_more_data"
            )
        
        # Calculate composite flakiness score
        weights = {
            'flip_rate': 0.35,
            'commit_variance': 0.25,
            'retry_success_rate': 0.20,
            'duration_cv': 0.10,
            'unique_error_count': 0.10
        }
        
        # Normalize and weight features
        flakiness_score = (
            features['flip_rate'] * weights['flip_rate'] +
            features['commit_variance'] * weights['commit_variance'] +
            features['retry_success_rate'] * weights['retry_success_rate'] +
            min(features['duration_cv'], 1.0) * weights['duration_cv'] +
            min(features['unique_error_count'] / 5, 1.0) * weights['unique_error_count']
        )
        
        # Determine primary pattern
        if features['flip_rate'] > 0.3:
            pattern = "timing_dependency"
            action = "add_explicit_waits_or_retry_logic"
        elif features['commit_variance'] > 0.5:
            pattern = "environment_dependency"
            action = "isolate_test_environment"
        elif features['retry_success_rate'] > 0.7:
            pattern = "resource_contention"
            action = "add_resource_locks_or_increase_timeouts"
        elif features['duration_cv'] > 0.5:
            pattern = "performance_variance"
            action = "mock_external_dependencies"
        else:
            pattern = "unknown"
            action = "manual_investigation"
        
        confidence = min(features['execution_count'] / 100, 1.0)
        
        return FlakinessPrediction(
            test_id=test_id,
            flakiness_score=min(flakiness_score, 1.0),
            confidence=confidence,
            primary_pattern=pattern,
            recommended_action=action
        )
