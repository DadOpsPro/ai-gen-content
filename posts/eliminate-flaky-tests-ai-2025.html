How to Implement Flaky Test Elimination with AI in 2025

Flaky tests are the silent productivity killers lurking in your CI/CD pipeline. They pass one moment, fail the next—with no code changes in between. In 2025, AI in Software Testing has matured to the point where you can systematically identify, diagnose, and eliminate flaky tests with unprecedented accuracy. This guide walks you through implementing a robust AI-powered flaky test elimination system.

Why This Matters

Flaky tests erode trust in your test suite. When developers see random failures, they start ignoring test results entirely—a dangerous habit that lets real bugs slip through to production.

The numbers are stark:

  • Engineering teams lose an average of 15-25% of their CI time rerunning flaky tests
  • A single flaky test costs approximately $500-$1,500 per month in wasted developer time
  • Teams with high flaky test rates deploy 40% slower than those with stable suites

    • Traditional approaches—manual investigation, quarantine lists, and blanket retries—treat symptoms rather than causes. AI-powered flaky test elimination changes the game by:

  • Automatically detecting flaky tests before they contaminate your main branch
  • Root-cause analysis that identifies why a test is flaky (timing, resource contention, test order dependency)
  • Intelligent remediation that suggests or auto-generates fixes

    flaky_detection/test_reporter.py
    
import json
import time
import os
import hashlib
from datetime import datetime
from dataclasses import dataclass, asdict
from typing import Optional

    

    @dataclass
class TestExecutionRecord:
    test_id: str
    test_name: str
    file_path: str
    outcome: str  # 'passed', 'failed', 'skipped'
    duration_ms: float
    timestamp: str
    run_id: str
    commit_sha: str
    branch: str
    retry_number: int
    error_message: Optional[str] = None
    error_type: Optional[str] = None
    environment: Optional[dict] = None

    

    class FlakynessDataCollector:
    def __init__(self, output_path: str = "./test_executions.jsonl"):
        self.output_path = output_path
        self.run_id = os.environ.get("CI_RUN_ID", self._generate_run_id())
        
    def _generate_run_id(self) -> str:
        return hashlib.md5(f"{datetime.now().isoformat()}".encode()).hexdigest()[:12]
    
    def record_test(self, test_name: str, file_path: str, outcome: str, 
                    duration_ms: float, retry_number: int = 0,
                    error_message: str = None, error_type: str = None):
        
        record = TestExecutionRecord(
            test_id=hashlib.md5(f"{file_path}::{test_name}".encode()).hexdigest(),
            test_name=test_name,
            file_path=file_path,
            outcome=outcome,
            duration_ms=duration_ms,
            timestamp=datetime.utcnow().isoformat(),
            run_id=self.run_id,
            commit_sha=os.environ.get("GIT_COMMIT", "unknown"),
            branch=os.environ.get("GIT_BRANCH", "unknown"),
            retry_number=retry_number,
            error_message=error_message,
            error_type=error_type,
            environment={
                "os": os.environ.get("RUNNER_OS", "unknown"),
                "python_version": os.environ.get("PYTHON_VERSION", "unknown"),
                "parallel_workers": os.environ.get("PYTEST_WORKERS", "1")
            }
        )
        
        with open(self.output_path, "a") as f:
            f.write(json.dumps(asdict(record)) + "\n")

    Step 2: Build the Flaky Test Detection Model

    With execution data flowing, implement an AI-based detection system. This model analyzes historical patterns to flag tests with flaky behavior. 

    flaky_detection/ai_detector.py
    
import pandas as pd
import numpy as np
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
from typing import List, Dict, Tuple
import json

    

    class AIFlakyTestDetector:
    def __init__(self, history_file: str):
        self.history_file = history_file
        self.model = IsolationForest(contamination=0.1, random_state=42)
        self.scaler = StandardScaler()
        
    def load_history(self) -> pd.DataFrame:
        records = []
        with open(self.history_file, "r") as f:
            for line in f:
                records.append(json.loads(line))
        return pd.DataFrame(records)
    
    def compute_flakiness_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Extract features that indicate flaky behavior."""
        
        features = df.groupby('test_id').agg({
            'outcome': [
                ('total_runs', 'count'),
                ('failure_rate', lambda x: (x == 'failed').mean()),
                ('pass_fail_transitions', lambda x: (x.shift() != x).sum())
            ],
            'duration_ms': [
                ('duration_mean', 'mean'),
                ('duration_std', 'std'),
                ('duration_cv', lambda x: x.std() / x.mean() if x.mean() > 0 else 0)
            ],
            'retry_number': [
                ('max_retries', 'max'),
                ('retry_rate', lambda x: (x > 0).mean())
            ]
        }).reset_index()
        
        features.columns = ['test_id'] + [col[1] for col in features.columns[1:]]
        
        # Calculate flip-flop score (passes then fails on same commit)
        flip_flop = df.groupby(['test_id', 'commit_sha'])['outcome'].apply(
            lambda x: x.nunique() > 1
        ).groupby('test_id').mean().reset_index()
        flip_flop.columns = ['test_id', 'flip_flop_rate']
        
        features = features.merge(flip_flop, on='test_id', how='left')
        features['flip_flop_rate'] = features['flip_flop_rate'].fillna(0)
        
        return features
    
    def calculate_flakiness_score(self, features: pd.DataFrame) -> pd.DataFrame:
        """Compute composite flakiness score using AI model."""
        
        feature_cols = ['failure_rate', 'pass_fail_transitions', 'duration_cv',
                       'retry_rate', 'flip_flop_rate']
        
        X = features[feature_cols].fillna(0)
        X_scaled = self.scaler.fit_transform(X)
        
        # Anomaly detection for unusual patterns
        anomaly_scores = -self.model.fit_predict(X_scaled)
        
        # Weighted composite score
        features['ai_flakiness_score'] = (
            features['flip_flop_rate'] * 0.35 +
            features['pass_fail_transitions'] / features['total_runs'] * 0.25 +
            features['duration_cv'].clip(upper=1) * 0.15 +
            features['retry_rate'] * 0.15 +
            (anomaly_scores / anomaly_scores.max()) * 0.10
        )
        
        features['is_flaky'] = features['ai_flakiness_score'] > 0.3
        
        return features.sort_values('ai_flakiness_score', ascending=False)
    
    def detect_flaky_tests(self) -> List[Dict]:
        """Main detection pipeline."""
        df = self.load_history()
        features = self.compute_flakiness_features(df)
        results = self.calculate_flakiness_score(features)
        
        flaky_tests = results[results['is_flaky']].to_dict('records')
        return flaky_tests

import os
from openai import OpenAI
from typing import Dict, List

    

    class FlakynessRootCauseAnalyzer:
    def __init__(self):
        self.client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
        
    def analyze_test(self, test_info: Dict, test_code: str, 
                     failure_samples: List[str]) -> Dict:
        """Use LLM to identify root cause of flakiness."""
        
        prompt = f"""Analyze this flaky test and identify the root cause.

    

    TEST INFORMATION:

  • Name: {test_info['test_name']}
    • 

  • Failure rate: {test_info['failure_rate']:.2%}
    • 

  • Duration variance (CV): {test_info['duration_cv']:.2f}
    • 

  • Flip-flop rate: {test_info['flip_flop_rate']:.2%}
    • 

    

    TEST CODE:
    python {test_code} 
    
SAMPLE ERROR MESSAGES FROM FAILURES:
{chr(10).join(failure_samples[:5])}

    

    Classify the flakiness into one of these categories:

  • TIMING_DEPENDENCY - Race conditions, async issues, timeouts
    • 

  • RESOURCE_CONTENTION - Shared state, database conflicts, port conflicts
    • 

  • TEST_ORDER_DEPENDENCY - Relies on other tests running first
    • 

  • ENVIRONMENT_SENSITIVITY - OS, timezone, locale differences
    • 

  • EXTERNAL_DEPENDENCY - Network calls, third-party services
    • 

  • NON_DETERMINISTIC_DATA - Random values, timestamps, UUIDs
    • 

    

    Respond in JSON format:
{{
    "category": "CATEGORY_NAME",
    "confidence": 0.0-1.0,
    "explanation": "Brief explanation",
    "code_locations": ["line numbers or snippets causing issues"],
    "fix_suggestion": "Specific fix recommendation"
}}"""

    

            response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        return json.loads(response.choices[0].message.content)
