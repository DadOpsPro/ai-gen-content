Why Long-Running AI Agents Break on HTTP and How Ably Is Fixing It

When your AI agent works flawlessly during development but mysteriously dies in production after 30 seconds, you've encountered one of the most frustrating problems in modern AI infrastructure. Long-running agents that perform complex reasoning, multi-step tasks, or real-time collaboration are fundamentally incompatible with traditional HTTP architecture—and most developers learn this the hard way.

This guide walks you through understanding why this break happens, how to diagnose it, and how to implement a robust fix using real-time messaging infrastructure.

Why This Matters: The HTTP Timeout Problem

Traditional HTTP was designed for quick request-response cycles: a browser asks for a webpage, the server returns it in milliseconds. AI agents operate on an entirely different timescale.

Consider a typical enterprise AI agent workflow:

  • User submits a complex analysis request
  • Agent reasons through the problem (5-10 seconds)
  • Agent calls multiple external APIs (15-30 seconds)
  • Agent processes and synthesizes results (10-20 seconds)
  • Agent generates and validates response (5-10 seconds)
    • Total time: 35-70+ seconds

    Here's where things break:

    • Load balancers typically timeout at 30-60 seconds
    • API gateways enforce strict request limits
    • Proxy servers close idle connections
    • Browser fetch APIs have default timeouts
    • Serverless functions have execution limits (AWS Lambda: 15 minutes, but HTTP response timeout is much shorter)

      • When any link in this chain breaks, your agent's work is lost, the user sees an error, and your reliability metrics plummet.

      The Real Cost of Connection Failures

      For AI agents in production, broken connections mean:

    • Lost computational work: GPU cycles already consumed
    • Inconsistent state: Partially completed tasks with no recovery path
    • Poor user experience: Users retry, multiplying load
    • Increased costs: Repeated API calls to LLM providers

    // ably-agent-channel.js
const Ably = require('ably');

    

    class AgentCommunicationChannel {
  constructor(apiKey) {
    this.ably = new Ably.Realtime({
      key: apiKey,
      // Critical: Enable automatic reconnection
      disconnectedRetryTimeout: 1000,
      suspendedRetryTimeout: 5000,
    });
    
    this.channels = new Map();
  }

    

      async createAgentSession(sessionId) {
    const channelName = agent:${sessionId};
    
    // Create channels for bidirectional communication
    const requestChannel = this.ably.channels.get(${channelName}:requests);
    const responseChannel = this.ably.channels.get(${channelName}:responses);
    const statusChannel = this.ably.channels.get(${channelName}:status);

    

        // Enable presence for connection state awareness
    await requestChannel.presence.enter({ role: 'orchestrator' });

    

        this.channels.set(sessionId, {
      requests: requestChannel,
      responses: responseChannel,
      status: statusChannel
    });

    

        return sessionId;
  }

    

      async sendToAgent(sessionId, task) {
    const channels = this.channels.get(sessionId);
    if (!channels) throw new Error('Session not found');

    

        const taskId = task_${Date.now()}_${Math.random().toString(36).substr(2, 9)};
    
    await channels.requests.publish('task', {
      taskId,
      payload: task,
      timestamp: Date.now()
    });

    

        return taskId;
  }

    

      subscribeToResponses(sessionId, callback) {
    const channels = this.channels.get(sessionId);
    if (!channels) throw new Error('Session not found');

    

        channels.responses.subscribe('result', (message) => {
      callback(null, message.data);
    });

    

        channels.responses.subscribe('error', (message) => {
      callback(new Error(message.data.error), null);
    });

    

        // Subscribe to incremental updates (streaming)
    channels.responses.subscribe('chunk', (message) => {
      callback(null, { type: 'chunk', data: message.data });
    });
  }

    

      subscribeToStatus(sessionId, callback) {
    const channels = this.channels.get(sessionId);
    if (!channels) throw new Error('Session not found');

    

        channels.status.subscribe((message) => {
      callback(message.data);
    });
  }
}

    

    module.exports = AgentCommunicationChannel;

    Step 3: Implement the Long-Running Agent Worker

    Create an agent worker that processes tasks independently of HTTP timeouts: 

    // agent-worker.js
const Ably = require('ably');
const OpenAI = require('openai');

    

    class LongRunningAgentWorker {
  constructor(ablyApiKey, openaiApiKey) {
    this.ably = new Ably.Realtime({ key: ablyApiKey });
    this.openai = new OpenAI({ apiKey: openaiApiKey });
    this.activeTasks = new Map();
  }

    

      async startListening(agentId) {
    const requestChannel = this.ably.channels.get(agent:${agentId}:requests);
    const responseChannel = this.ably.channels.get(agent:${agentId}:responses);
    const statusChannel = this.ably.channels.get(agent:${agentId}:status);

    

        // Enter presence so clients know we're available
    await requestChannel.presence.enter({ role: 'worker', status: 'ready' });

    

        requestChannel.subscribe('task', async (message) => {
      const { taskId, payload } = message.data;
      
      console.log(Received task: ${taskId});
      
      // Track active task for potential recovery
      this.activeTasks.set(taskId, {
        startTime: Date.now(),
        status: 'processing'
      });

    

          try {
        // Publish status update
        await statusChannel.publish('update', {
          taskId,
          status: 'started',
          timestamp: Date.now()
        });

    

            // Process with streaming for long operations
        await this.processTaskWithStreaming(
          taskId,
          payload,
          responseChannel,
          statusChannel
        );

    

          } catch (error) {
        console.error(Task ${taskId} failed:, error);
        
        await responseChannel.publish('error', {
          taskId,
          error: error.message,
          recoverable: this.isRecoverableError(error)
        });
      } finally {
        this.activeTasks.delete(taskId);
      }
    });

    

        console.log(Agent worker listening on channel: agent:${agentId});
  }

    

      async processTaskWithStreaming(taskId, payload, responseChannel, statusChannel) {
    const { prompt, context, tools } = payload;

    

        // For long-running tasks, stream intermediate results
    const stream = await this.openai.chat.completions.create({
      model: 'gpt-4',
      messages: [
        { role: 'system', content: context || 'You are a helpful assistant.' },
        { role: 'user', content: prompt }
      ],
      stream: true,
      // Enable tool use for agentic workflows
      tools: tools || undefined
    });

    

        let fullResponse = '';
    let chunkCount = 0;

    

        for await (const chunk of stream) {
      const content = chunk.choices[0]?.delta?.content || '';
      fullResponse += content;
      chunkCount++;

    

          // Send chunks in batches to reduce message overhead
      if (chunkCount % 10 === 0 && content) {
        await responseChannel.publish('chunk', {
          taskId,
          content: fullResponse,
          isPartial: true
        });

    

            // Update status with progress
        await statusChannel.publish('update', {
          taskId,
          status: 'streaming',
          progress: ${fullResponse.length} characters generated
        });
      }
    }

    

        // Send final result
    await responseChannel.publish('result', {
      taskId,
      content: fullResponse,
      isPartial: false,
      completedAt: Date.now()
    });

    

        await statusChannel.publish('update', {
      taskId,
      status: 'completed',
      timestamp: Date.now()
    });
  }

    

      isRecoverableError(error) {
    const recoverablePatterns = [
      'rate_limit',
      'timeout',
      'ECONNRESET',
      'socket hang up'
    ];
    return recoverablePatterns.some(pattern => 
      error.message.toLowerCase().includes(pattern.toLowerCase())
    );
  }
}

    Step 4: Build the Client-Side Integration

    Create a client that gracefully handles long-running operations: 

    // client-integration.js
const AgentCommunicationChannel = require('./ably-agent-channel');

    

    class AIAgentClient {
  constructor(ablyApiKey) {
    this.channel = new AgentCommunicationChannel(ablyApiKey);
    this.pendingTasks = new Map();
  }

    

      async submitTask(prompt, options = {}) {
    const sessionId = options.sessionId || this.generateSessionId();
    
    // Initialize session if new
    if (!this.channel.channels.has(sessionId)) {
      await this.channel.createAgentSession(sessionId);
    }

    

        return new Promise((resolve, reject) => {
      const timeout = options.timeout || 300000; // 5 minute default
      let resolved = false;
      let accumulatedContent = '';

    

          // Set up timeout (but much longer than HTTP allows)
      const timeoutId = setTimeout(() => {
        if (!resolved) {
          resolved = true;
          reject(new Error(Task timed out after ${timeout}ms));
        }
      }, timeout);

    

          // Subscribe to responses
      this.channel.subscribeToResponses(sessionId, (error, data) => {
        if (resolved) return;

    

            if (error) {
          resolved = true;
          clearTimeout(timeoutId);
          reject(error);
          return;
        }

    

            if (data.type === 'chunk') {
          // Handle streaming updates
          accumulatedContent = data.data.content;
          if (options.onChunk) {
            options.onChunk(accumulatedContent);
          }
        } else if (!data.isPartial) {
          // Final result
          resolved = true;
          clearTimeout(timeoutId);
          resolve({
            content: data.content,
            taskId: data.taskId,
            completedAt: data.completedAt
          });
        }
      });

    

          // Subscribe to status for UI updates
      if (options.onStatus) {
        this.channel.subscribeToStatus(sessionId, options.onStatus);
      }

    

          // Submit the task
      this.channel.sendToAgent(sessionId, {
        prompt,
        context: options.context,
        tools: options.tools
      }).catch(reject);
    });
  }

    

      generateSessionId() {
    return session_${Date.now()}_${Math.random().toString(36).substr(2, 9)};
  }
}
