Implement a boundary enforcement layer that validates every agent action: 

boundary_enforcer.py

from dataclasses import dataclass
from typing import List, Set
import yaml
import logging



@dataclass
class AgentBoundaries:
    allowed_actions: Set[str]
    prohibited_actions: Set[str]
    max_api_calls: int
    max_execution_time: int
    
class BoundaryEnforcer:
    def __init__(self, config_path: str):
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        self.boundaries = AgentBoundaries(
            allowed_actions=set(config['boundaries']['allowed_actions']),
            prohibited_actions=set(config['boundaries']['prohibited_actions']),
            max_api_calls=config['boundaries']['resource_limits']['max_api_calls_per_request'],
            max_execution_time=config['boundaries']['resource_limits']['max_execution_time_seconds']
        )
        self.api_call_count = 0
        self.logger = logging.getLogger(__name__)
    
    def validate_action(self, action: str, context: dict) -> tuple[bool, str]:
        """
        Validates if an agent action is permitted.
        Returns (is_valid, reason)
        """
        # Check explicit prohibitions first
        if action in self.boundaries.prohibited_actions:
            self.logger.warning(f"Blocked prohibited action: {action}")
            return False, f"Action '{action}' is explicitly prohibited"
        
        # Check allowlist
        if action not in self.boundaries.allowed_actions:
            self.logger.warning(f"Blocked unlisted action: {action}")
            return False, f"Action '{action}' is not in allowed actions list"
        
        # Check resource limits
        self.api_call_count += 1
        if self.api_call_count > self.boundaries.max_api_calls:
            return False, "Maximum API calls exceeded for this request"
        
        return True, "Action permitted"
    
    def reset_session(self):
        """Reset counters for new agent session"""
        self.api_call_count = 0

class AgentObserver:
    def __init__(self, agent_id: str):
        self.agent_id = agent_id
        self.logger = structlog.get_logger()
        self.tracer = trace.get_tracer(__name__)
    
    def trace_agent_execution(self, request_id: str):
        """Context manager for tracing full agent execution"""
        return self.tracer.start_as_current_span(
            "agent_execution",
            attributes={
                "agent.id": self.agent_id,
                "request.id": request_id
            }
        )
    
    def log_decision(self, 
                     request_id: str,
                     decision_type: str,
                     input_context: dict,
                     reasoning: str,
                     chosen_action: str,
                     confidence: float,
                     alternatives_considered: list):
        """
        Log every decision point for audit and debugging.
        This is crucial for understanding agent behavior in production.
        """
        self.logger.info(
            "agent_decision",
            request_id=request_id,
            agent_id=self.agent_id,
            decision_type=decision_type,
            input_summary=self._summarize_input(input_context),
            reasoning=reasoning,
            chosen_action=chosen_action,
            confidence=confidence,
            alternatives=alternatives_considered,
            timestamp=datetime.utcnow().isoformat()
        )
    
    def log_tool_call(self,
                      request_id: str,
                      tool_name: str,
                      parameters: dict,
                      result: dict,
                      latency_ms: float,
                      success: bool):
        """Track every external tool/API call the agent makes"""
        with self.tracer.start_as_current_span("tool_call") as span:
            span.set_attribute("tool.name", tool_name)
            span.set_attribute("tool.success", success)
            span.set_attribute("tool.latency_ms", latency_ms)
            
            if not success:
                span.set_status(Status(StatusCode.ERROR))
            
            self.logger.info(
                "agent_tool_call",
                request_id=request_id,
                tool_name=tool_name,
                parameters=self._sanitize_params(parameters),
                result_summary=self._summarize_result(result),
                latency_ms=latency_ms,
                success=success
            )
    
    def _summarize_input(self, context: dict) -> dict:
        """Summarize input to avoid logging sensitive data"""
        return {
            "keys": list(context.keys()),
            "size_bytes": len(json.dumps(context))
        }
    
    def _sanitize_params(self, params: dict) -> dict:
        """Remove sensitive parameters before logging"""
        sensitive_keys = {'password', 'token', 'secret', 'ssn', 'credit_card'}
        return {k: '[REDACTED]' if k.lower() in sensitive_keys else v 
                for k, v in params.items()}
    
    def _summarize_result(self, result: dict) -> dict:
        """Summarize results for logging"""
        return {
            "status": result.get("status"),
            "has_data": bool(result.get("data")),
            "error": result.get("error")
        }

Step 3: Build Circuit Breakers and Fallback Mechanisms

Production agents must degrade gracefully. Implement circuit breakers that activate when agent behavior becomes erratic: 

circuit_breaker.py

from enum import Enum
from datetime import datetime, timedelta
from collections import deque
import threading



class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject requests
    HALF_OPEN = "half_open"  # Testing recovery



class AgentCircuitBreaker:
    def __init__(self,
                 failure_threshold: int = 5,
                 success_threshold: int = 3,
                 timeout_seconds: int = 60,
                 window_seconds: int = 120):
        self.failure_threshold = failure_threshold
        self.success_threshold = success_threshold
        self.timeout_seconds = timeout_seconds
        self.window_seconds = window_seconds
        
        self.state = CircuitState.CLOSED
        self.failures = deque()
        self.successes_in_half_open = 0
        self.last_failure_time = None
        self.lock = threading.Lock()
    
    def can_execute(self) -> tuple[bool, str]:
        """Check if agent execution should proceed"""
        with self.lock:
            self._clean_old_failures()
            
            if self.state == CircuitState.CLOSED:
                return True, "Circuit closed, proceeding"
            
            if self.state == CircuitState.OPEN:
                if self._should_attempt_reset():
                    self.state = CircuitState.HALF_OPEN
                    return True, "Circuit half-open, testing"
                return False, "Circuit open, request rejected"
            
            if self.state == CircuitState.HALF_OPEN:
                return True, "Circuit half-open, testing"
        
        return False, "Unknown state"
    
    def record_success(self):
        """Record successful agent execution"""
        with self.lock:
            if self.state == CircuitState.HALF_OPEN:
                self.successes_in_half_open += 1
                if self.successes_in_half_open >= self.success_threshold:
                    self.state = CircuitState.CLOSED
                    self.successes_in_half_open = 0
                    self.failures.clear()
    
    def record_failure(self, error_type: str):
        """Record failed agent execution"""
        with self.lock:
            now = datetime.utcnow()
            self.failures.append((now, error_type))
            self.last_failure_time = now
            
            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.OPEN
                self.successes_in_half_open = 0
            elif len(self.failures) >= self.failure_threshold:
                self.state = CircuitState.OPEN
    
    def _clean_old_failures(self):
        """Remove failures outside the time window"""
        cutoff = datetime.utcnow() - timedelta(seconds=self.window_seconds)
        while self.failures and self.failures[0][0] < cutoff:
            self.failures.popleft()
    
    def _should_attempt_reset(self) -> bool:
        """Check if enough time has passed to test recovery"""
        if self.last_failure_time is None:
            return True
        return datetime.utcnow() > self.last_failure_time + timedelta(seconds=self.timeout_seconds)
